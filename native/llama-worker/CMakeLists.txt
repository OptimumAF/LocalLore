function(rebase_output PREFIX VALUE RESULT)
  if(VALUE STREQUAL "" OR VALUE MATCHES "^:" OR VALUE MATCHES "[;\r\n]")
    message(FATAL_ERROR "Unsupported worker compiler output")
  endif()
  string(REPLACE "\\" "/" VALUE "${VALUE}")
  get_filename_component(OUTPUT "${VALUE}" ABSOLUTE BASE_DIR "${BINARY_DIR}")
  string(TOLOWER "${OUTPUT}" OUTPUT_ID)
  string(FIND "${OUTPUT_ID}" "${BINARY_ID}/" PREFIX_AT)
  if(NOT PREFIX_AT EQUAL 0)
    message(FATAL_ERROR "Worker compiler output escaped build directory")
  endif()
  if(PREFIX STREQUAL "/Fo" AND NOT OUTPUT MATCHES "\\.obj$")
    message(FATAL_ERROR "Unsupported worker object output")
  endif()
  set(PDB_DIRECTORY FALSE)
  if(PREFIX STREQUAL "/Fd" AND VALUE MATCHES "/$" AND IS_DIRECTORY "${OUTPUT}")
    set(PDB_DIRECTORY TRUE)
    string(APPEND OUTPUT "/")
  endif()
  if(PREFIX STREQUAL "/Fd" AND NOT OUTPUT MATCHES "\\.pdb$" AND
      NOT OUTPUT STREQUAL "${BINARY_DIR}/TARGET_COMPILE_PDB" AND NOT PDB_DIRECTORY)
    message(FATAL_ERROR "Unsupported worker PDB output")
  endif()
  set(${RESULT} "${PREFIX}${OUTPUT}" PARENT_SCOPE)
endfunction()
